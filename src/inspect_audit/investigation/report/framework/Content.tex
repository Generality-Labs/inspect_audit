% EXAMPLE --- FULL REPORT CONTENT SKELETON (one file, in reading order)
%
% To start a new audit: copy reports/Example/ to reports/<Eval>/, rename
% Example_Audit.tex to <Eval>_Audit.tex, fill in its reference numbers,
% then replace every [bracketed placeholder] in this file. Compile from
% inside the report directory (its .latexmkrc finds the shared class,
% style and sections in template/).
%
% Structure per findings dimension:
%   Overall Assessment and summary
%   Contribution Assessment table (every registry contribution)
%   Evidence for Observed Concerns (only where concerns are identified)
% Replace [assessment] with the appropriate assessment macro, including
% \assessNotAssessed or \assessNA. Add evidence links to relevant rows.
% The evidence blocks below are placeholders: remove them for dimensions
% without observed concerns, or duplicate them for additional concerns.
% ==================================================================

% --- DIMENSION ASSESSMENTS -----------------------------------------
% Drives BOTH the scorecard table and the findings boxes. Assessment
% levels: \assessNone \assessMinor \assessMajor \assessCritical
% \assessNotAssessed \assessNA. Key contributions use
% \issue{code}{name}{one-line description}.
\setdimension{construct}{[assessment]}{\issue{A.1}{[contribution]}{[one-line description]}}
\setdimension{contentvalidity}{[assessment]}{[key contributions]}
\setdimension{dataset}{[assessment]}{[key contributions]}
\setdimension{scaffold}{[assessment]}{[key contributions]}
\setdimension{harness}{[assessment]}{[key contributions]}
\setdimension{environment}{[assessment]}{[key contributions]}
\setdimension{grading}{[assessment]}{[key contributions]}
\setdimension{resources}{[assessment]}{[key contributions]}
\setdimension{informativeness}{[assessment]}{[key contributions]}
% Scorecard rows, most to least critical; delete a key to omit its row.
\scorecardorder{construct,contentvalidity,dataset,scaffold,harness,environment,grading,resources,informativeness}
